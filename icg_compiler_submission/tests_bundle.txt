    else

int square(int x) { return x * x; }

int apply(BinOp f, int x, int y) { return f(x, y); }

BinOp pick(int which)

    if (which == 2) return sub;

}

}

    ops[2] = mul;

    bag.second = mul;

{
    va_list ap;
    int i;
    double total = 0;

        total += va_arg(ap, double);
    va_end(ap);

int maxOf(int first, ...)
{
    va_list ap;
    int best = first, v;
    va_start(ap, first);
    while ((v = va_arg(ap, int)) != -9999)
        if (v > best) best = v;
    va_end(ap);
    return best;
}

    int i;

        printf("(%d,", va_arg(ap, int));
        printf("%f) ", va_arg(ap, double));

    va_end(ap);
}

    va_list ap;

int writeText(const char *path)
{
    FILE *f = fopen(path, "w");
    int i;
    if (f == NULL) return -1;
    fprintf(f, "line %d\n", 1);
    fputs("plain text\n", f);
    fputc('X', f);
    fputc('\n', f);

        fprintf(f, "%d %f\n", i, i * 0.5);
    fclose(f);

int readText(const char *path)
{
    FILE *f = fopen(path, "r");
    char buf[128];
    int n = 0, sum = 0, v;
    double d;
    if (f == NULL) return -1;
    while (fgets(buf, sizeof(buf), f) != NULL)
        n++;
    rewind(f);
    while (fscanf(f, "%d %lf", &v, &d) == 2)
        sum += v;
    while (!feof(f)) {
        if (fgetc(f) == EOF) break;
        sum++;
    }
    fclose(f);
    return sum + n * 100;
}

int binaryIO(const char *path)
{
    FILE *f = fopen(path, "wb");
    int data[4] = {10, 20, 30, 40};
    int back[4] = {0, 0, 0, 0};
    long pos;
    int i, ok = 1;
    if (f == NULL) return -1;
    fwrite(data, sizeof(int), 4, f);
    fclose(f);

    f = fopen(path, "rb");
    if (f == NULL) return -1;
    pos = ftell(f);
    fseek(f, 2 * sizeof(int), 0);      /* SEEK_SET == 0 */
    fread(back, sizeof(int), 2, f);
    fclose(f);
    if (back[0] != 30 || back[1] != 40) ok = 0;
    return ok;
}

// Lambdas: no capture, capture-by-value, capture-by-reference, named
// captures, return-type annotations, calling immediately, and lambdas
// that capture locals in nested scopes.
#include <stdio.h>

// EXPECT-IR: func main() : int
// EXPECT-IR: nothing = __lambda1
// EXPECT-IR: typed = __lambda4
// EXPECT-IR: namedRef = __lambda8
// EXPECT-IR: capAllVal = __lambda11
// EXPECT-IR: func __lambda1(

int apply2(int (*f)(int, int), int a, int b)
{
    return f(a, b);
}

int main()
{
    int base = 10;
    int offset = 5;

    auto nothing = []() { printf("no capture\n"); };
    auto square = [](int v) { return v * v; };
    auto add = [](int a, int b) { return a + b; };
    auto typed = [](double d) -> double { return d / 2; };

    auto byValue = [=](int v) { return v + base + offset; };   /* [=] captures both */
    auto byRef = [&](int v) { base += v; offset += v; return base + offset; };
    auto named = [base](int v) { return v * base; };           /* named by-value */
    auto namedRef = [&offset](int v) { offset = v; return offset; };
    auto mixed = [&base, offset](int v) { return base + offset + v; };

    auto capAllRef = [&]() { base = base * 2; };
    auto capAllVal = [=]() { return base + offset; };

    nothing();
    printf("%d\n", square(6));            // 36
    printf("%d\n", add(20, 22));          // 42
    printf("%f\n", typed(9.0));           // 4.5

    printf("%d\n", byValue(1));           // 16
    printf("%d\n", byRef(1));             // 17 (base=11, offset=6)
    printf("%d %d\n", base, offset);      // 11 6
    printf("%d\n", named(3));             // 33
    printf("%d\n", namedRef(99));         // 99
    printf("%d\n", offset);               // 99
    printf("%d\n", mixed(1));             // 11 + 99 + 1 = 111

    capAllRef();
    printf("%d\n", base);                 // 22
    printf("%d\n", capAllVal());          // 121

    printf("%d\n", apply2(add, 3, 4));    // 7 (lambda converts to function pointer)
    printf("%d\n", ([](int x) { return x + 1; })(41));   // 42 (immediately invoked)

    {   /* captures of variables in nested scopes */
        int inner = 7;
        auto grab = [&inner](int v) { inner += v; return inner; };
        printf("%d\n", grab(3));         // 10
        printf("%d\n", inner);           // 10
    }
    return 0;
}

    int r2 = 2 + 3 * 4;              // EXPECT-IR: r2 = 14

    int r4 = 7 + 9 % 4;              // EXPECT-IR: r4 = 8

    int r6 = 1 << 2 + 3;             // EXPECT-IR: r6 = 32
    int r7 = 16 >> 1 + 1;            // EXPECT-IR: r7 = 4
    int r8 = (1 << 3) - 1;           // EXPECT-IR: r8 = 7

    int r9 = 1 < 2 && 3 < 4 || 0;    // EXPECT-IR: r9 = 1

    int x = 5;

    return isOdd(n - 1);

{
    if (m == 0) return n + 1;
    if (n == 0) return ack(m - 1, 1);

}

{
    static int memo[40];
    if (n < 2) return n;
    if (memo[n] != 0) return memo[n];
    memo[n] = fibm(n - 1) + fibm(n - 2);
    return memo[n];
